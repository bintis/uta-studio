cmake_minimum_required(VERSION 3.20)
project(uta_libtorch_native LANGUAGES CXX)

set(TORCH_ROOT "" CACHE PATH "Explicit native LibTorch root containing include and lib")
set(ROCM_ROOT "" CACHE PATH "Explicit expanded private ROCm SDK root")
set(ROCM_CORE_ROOT "" CACHE PATH "Optional private ROCm core runtime root; defaults beside ROCM_ROOT")
set(ROCM_LIBRARIES_ROOT "" CACHE PATH "Optional private ROCm libraries runtime root; defaults beside ROCM_ROOT")
set(XPU_DEPENDENCY_LIB "" CACHE PATH "Explicit private XPU wheel dependency library directory")
set(UTA_LIBTORCH_BACKEND "cpu" CACHE STRING "Native build backend: cpu, rocm or xpu")
set_property(CACHE UTA_LIBTORCH_BACKEND PROPERTY STRINGS cpu rocm xpu)
set(TORCH_CXX_ABI "1" CACHE STRING "LibTorch libstdc++ ABI selection")
if(NOT EXISTS "${TORCH_ROOT}/include/ATen/ATen.h")
  message(FATAL_ERROR "TORCH_ROOT must identify the selected native runtime; dependencies are not downloaded by this build")
endif()

add_library(uta_libtorch SHARED
  api.cpp gguf.cpp runtime.cpp roformer.cpp rmvpe.cpp fcpe.cpp basic_pitch.cpp jbm.cpp
  game.cpp qwen.cpp firered.cpp note_network.cpp stars.cpp rosvot.cpp)
target_compile_features(uta_libtorch PUBLIC cxx_std_20)
target_compile_options(uta_libtorch PRIVATE -Wall -Wextra)
target_compile_definitions(uta_libtorch PRIVATE _GLIBCXX_USE_CXX11_ABI=${TORCH_CXX_ABI})
target_include_directories(uta_libtorch SYSTEM PRIVATE
  "${TORCH_ROOT}/include" "${TORCH_ROOT}/include/torch/csrc/api/include")
target_include_directories(uta_libtorch PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>")
set(native_components torch torch_cpu c10)
set(native_paths "${TORCH_ROOT}/lib")
if(UTA_LIBTORCH_BACKEND STREQUAL "rocm")
  if(NOT EXISTS "${ROCM_ROOT}/include/hip/hip_runtime_api.h")
    message(FATAL_ERROR "ROCM_ROOT is required for the explicitly selected ROCm build")
  endif()
  if(ROCM_CORE_ROOT STREQUAL "")
    get_filename_component(_rocm_site_packages "${ROCM_ROOT}" DIRECTORY)
    if(EXISTS "${_rocm_site_packages}/_rocm_sdk_core/lib")
      set(ROCM_CORE_ROOT "${_rocm_site_packages}/_rocm_sdk_core")
    endif()
  endif()
  if(ROCM_LIBRARIES_ROOT STREQUAL "")
    get_filename_component(_rocm_site_packages "${ROCM_ROOT}" DIRECTORY)
    if(EXISTS "${_rocm_site_packages}/_rocm_sdk_libraries/lib")
      set(ROCM_LIBRARIES_ROOT "${_rocm_site_packages}/_rocm_sdk_libraries")
    endif()
  endif()
  list(APPEND native_components torch_hip c10_hip)
  target_compile_definitions(uta_libtorch PRIVATE UTA_LIBTORCH_ROCM=1 __HIP_PLATFORM_AMD__=1)
  target_include_directories(uta_libtorch SYSTEM PRIVATE "${ROCM_ROOT}/include")
  find_library(native_hip NAMES amdhip64 PATHS "${ROCM_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
  find_library(native_host_blas NAMES rocm-openblas librocm-openblas.so.0
    PATHS "${ROCM_ROOT}/lib/host-math/lib" NO_DEFAULT_PATH REQUIRED)
  # AOTriton requires the standard XZ ABI, not ROCm's renamed private symbols.
  find_library(native_lzma NAMES lzma REQUIRED)
  target_link_libraries(uta_libtorch PRIVATE "-Wl,--no-as-needed"
    "${native_hip}" "${native_host_blas}" "${native_lzma}" "-Wl,--as-needed")
  list(APPEND native_paths "${ROCM_ROOT}/lib" "${ROCM_ROOT}/lib/host-math/lib" "${ROCM_ROOT}/lib/rocm_sysdeps/lib" "${ROCM_ROOT}/lib/llvm/lib")
  if(EXISTS "${ROCM_CORE_ROOT}/lib")
    list(APPEND native_paths "${ROCM_CORE_ROOT}/lib")
  endif()
  if(EXISTS "${ROCM_LIBRARIES_ROOT}/lib")
    list(APPEND native_paths "${ROCM_LIBRARIES_ROOT}/lib")
  endif()
  # PyTorch's ROCm wheel dependencies live in sibling package roots. GNU ld
  # RUNPATH is not inherited by transitive dependencies, so use an inherited
  # private RPATH for this isolated runtime instead of requiring global loader state.
  target_link_options(uta_libtorch PRIVATE "-Wl,--disable-new-dtags")
elseif(UTA_LIBTORCH_BACKEND STREQUAL "xpu")
  if(XPU_DEPENDENCY_LIB STREQUAL "" OR NOT EXISTS "${XPU_DEPENDENCY_LIB}")
    message(FATAL_ERROR "XPU_DEPENDENCY_LIB is required for the explicitly selected XPU build")
  endif()
  list(APPEND native_components torch_xpu c10_xpu)
  list(APPEND native_paths "${XPU_DEPENDENCY_LIB}")
  target_compile_definitions(uta_libtorch PRIVATE UTA_LIBTORCH_XPU=1)
  # torch_xpu and SYCL have indirect dependencies in the private wheel root.
  # RUNPATH on this library alone is not inherited by those dependencies.
  target_link_options(uta_libtorch PRIVATE "-Wl,--disable-new-dtags" "-Wl,-rpath-link,${XPU_DEPENDENCY_LIB}")
elseif(NOT UTA_LIBTORCH_BACKEND STREQUAL "cpu")
  message(FATAL_ERROR "UTA_LIBTORCH_BACKEND must be cpu, rocm or xpu")
endif()
foreach(component IN LISTS native_components)
  find_library(native_${component} NAMES ${component} PATHS "${TORCH_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
  target_link_libraries(uta_libtorch PRIVATE "-Wl,--no-as-needed" "${native_${component}}" "-Wl,--as-needed")
endforeach()
set_target_properties(uta_libtorch PROPERTIES
  CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES BUILD_RPATH "${native_paths}")

add_executable(uta-libtorch-contract-check contract_check.cpp)
target_compile_features(uta-libtorch-contract-check PRIVATE cxx_std_20)
target_link_libraries(uta-libtorch-contract-check PRIVATE uta_libtorch)
if(UTA_LIBTORCH_BACKEND STREQUAL "rocm" OR UTA_LIBTORCH_BACKEND STREQUAL "xpu")
  target_link_options(uta-libtorch-contract-check PRIVATE "-Wl,--disable-new-dtags")
endif()
set_target_properties(uta-libtorch-contract-check PROPERTIES BUILD_RPATH "$<TARGET_FILE_DIR:uta_libtorch>;${native_paths}")

# GPU sequence parity check. Its CPU computation is an explicit independent
# double-precision oracle, never a production inference fallback.
add_executable(uta-libtorch-recurrent-check recurrent_check.cpp)
target_compile_features(uta-libtorch-recurrent-check PRIVATE cxx_std_20)
target_compile_definitions(uta-libtorch-recurrent-check PRIVATE _GLIBCXX_USE_CXX11_ABI=${TORCH_CXX_ABI})
target_include_directories(uta-libtorch-recurrent-check SYSTEM PRIVATE
  "${TORCH_ROOT}/include" "${TORCH_ROOT}/include/torch/csrc/api/include")
target_link_libraries(uta-libtorch-recurrent-check PRIVATE uta_libtorch)
foreach(component IN LISTS native_components)
  target_link_libraries(uta-libtorch-recurrent-check PRIVATE "-Wl,--no-as-needed" "${native_${component}}" "-Wl,--as-needed")
endforeach()
if(UTA_LIBTORCH_BACKEND STREQUAL "rocm")
  target_link_options(uta-libtorch-recurrent-check PRIVATE "-Wl,--disable-new-dtags")
endif()
set_target_properties(uta-libtorch-recurrent-check PROPERTIES BUILD_RPATH "$<TARGET_FILE_DIR:uta_libtorch>;${native_paths}")
install(TARGETS uta_libtorch LIBRARY DESTINATION lib)
install(FILES api.h DESTINATION include/uta_libtorch)
